    // Initial Confetti Burst
    function blastConfetti() {
      confetti({
        particleCount: 180,
        spread: 120,
        origin: { y: 0.6 }
      });
    }

    // Auto Confetti on Load
    window.onload = () => {
      blastConfetti();

      // Continuous small bursts
      setInterval(() => {
        confetti({
          particleCount: 40,
          spread: 70,
          origin: {
            x: Math.random(),
            y: Math.random() - 0.2
          }
        });
      }, 2500);
